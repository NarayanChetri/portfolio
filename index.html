First Come First Serve Non-Preemptive 
#include <iostream>
using namespace std;
int main() {
    int n;
    float avgwt = 0, avgtat = 0;
    cout << "Enter number of processes: ";
    cin >> n;
    int at[10], bt[10], wt[10], tat[10], ct[10], p[10];
    for(int i=0; i<n; i++) {
        p[i] = i + 1;
        cout << "\nP" << p[i] << endl;
        cout << "AT: "; cin >> at[i];
        cout << "BT: "; cin >> bt[i];
    }
    // Sort by Arrival Time
    for(int i=0; i<n-1; i++)
        for(int j=i+1; j<n; j++)
            if(at[i] > at[j]) {
                swap(at[i],at[j]); swap(bt[i],bt[j]); swap(p[i],p[j]);
            }
    // FCFS Calculation
    ct[0] = at[0] + bt[0];
    tat[0] = ct[0] - at[0];
    wt[0] = tat[0] - bt[0];
    for(int i=1; i<n; i++) {
        ct[i]  = (ct[i-1] < at[i]) ? at[i]+bt[i] : ct[i-1]+bt[i];
        tat[i] = ct[i] - at[i];
        wt[i]  = tat[i] - bt[i];
    }
    // Re-sort by Process Number
    for(int i=0; i<n-1; i++)
        for(int j=i+1; j<n; j++)
            if(p[i] > p[j]) {
                swap(p[i],p[j]); swap(at[i],at[j]); swap(bt[i],bt[j]);
                swap(wt[i],wt[j]); swap(tat[i],tat[j]);
            }
    cout << "\nP\tAT\tBT\tWT\tTAT\n";
    for(int i=0; i<n; i++) {
        cout << "P"<<p[i]<<"\t"<<at[i]<<"\t"<<bt[i]<<"\t"<<wt[i]<<"\t"<<tat[i]<<"\n";
        avgwt += wt[i]; avgtat += tat[i];
    }
    cout << "\nAverage WT  = " << avgwt/n;
    cout << "\nAverage TAT = " << avgtat/n;
    return 0;
}

#include <iostream>
using namespace std;
int main() {
    int n, at[10], bt[10], wt[10], tat[10], ct[10], p[10];
    bool done[10] = {};
    float avgwt=0, avgtat=0;
    cout << "Enter number of processes: "; cin >> n;
    for(int i=0; i<n; i++) {
        p[i]=i+1;
        cout << "\nP"<<p[i]<<" AT: "; cin >> at[i];
        cout << "P"<<p[i]<<" BT: "; cin >> bt[i];
    }
    int time=0, completed=0;
    while(completed < n) {
        int idx=-1, minbt=9999;
        for(int i=0; i<n; i++)
            if(!done[i] && at[i]<=time && bt[i]<minbt)
                { minbt=bt[i]; idx=i; }
        if(idx==-1) { time++; continue; }
        time += bt[idx];
        ct[idx]  = time;
        tat[idx] = ct[idx] - at[idx];
        wt[idx]  = tat[idx] - bt[idx];
        done[idx]=true; completed++;
    }
    cout << "\nP\tAT\tBT\tWT\tTAT\n";
    for(int i=0; i<n; i++) {
        cout << "P"<<p[i]<<"\t"<<at[i]<<"\t"<<bt[i]<<"\t"<<wt[i]<<"\t"<<tat[i]<<"\n";
        avgwt+=wt[i]; avgtat+=tat[i];
    }
    cout << "\nAverage WT  = " << avgwt/n;
    cout << "\nAverage TAT = " << avgtat/n;
    return 0;
}

#include <iostream>
using namespace std;
int main() {
    int n, at[10], bt[10], rem[10], ct[10], wt[10], tat[10], p[10];
    bool done[10]={};
    float avgwt=0, avgtat=0;
    cout << "Enter number of processes: "; cin >> n;
    for(int i=0; i<n; i++) {
        p[i]=i+1;
        cout << "\nP"<<p[i]<<" AT: "; cin >> at[i];
        cout << "P"<<p[i]<<" BT: "; cin >> bt[i];
        rem[i]=bt[i];
    }
    int time=0, completed=0;
    while(completed < n) {
        int idx=-1, minrem=9999;
        for(int i=0; i<n; i++)
            if(!done[i] && at[i]<=time && rem[i]<minrem)
                { minrem=rem[i]; idx=i; }
        if(idx==-1) { time++; continue; }
        rem[idx]--; time++;
        if(rem[idx]==0) {
            ct[idx]=time;
            tat[idx]=ct[idx]-at[idx];
            wt[idx]=tat[idx]-bt[idx];
            done[idx]=true; completed++;
        }
    }
    cout << "\nP\tAT\tBT\tWT\tTAT\n";
    for(int i=0; i<n; i++) {
        cout << "P"<<p[i]<<"\t"<<at[i]<<"\t"<<bt[i]<<"\t"<<wt[i]<<"\t"<<tat[i]<<"\n";
        avgwt+=wt[i]; avgtat+=tat[i];
    }
    cout << "\nAverage WT  = " << avgwt/n;

}

#include <iostream>
using namespace std;
int main() {
    int n, tq, at[10], bt[10], rem[10], ct[10], wt[10], tat[10], p[10];
    float avgwt=0, avgtat=0;
    cout << "Enter number of processes: "; cin >> n;
    cout << "Time Quantum: ";              cin >> tq;
    for(int i=0; i<n; i++) {
        p[i]=i+1;
        cout << "\nP"<<p[i]<<" AT: "; cin >> at[i];
        cout << "P"<<p[i]<<" BT: "; cin >> bt[i];
        rem[i]=bt[i];
    }
    // Sort by arrival time
    for(int i=0; i<n-1; i++)
        for(int j=i+1; j<n; j++)
            if(at[i]>at[j]) {
                swap(at[i],at[j]); swap(bt[i],bt[j]);
                swap(rem[i],rem[j]); swap(p[i],p[j]);
            }
    int queue[50], front=0, rear=0, inQ[10]={};
    int time=0, completed=0;
    queue[rear++]=0; inQ[0]=1;
    while(completed < n) {
        if(front==rear) { time++; // idle
            for(int i=0; i<n; i++)
                if(!inQ[i] && rem[i]>0 && at[i]<=time)
                    { queue[rear++]=i; inQ[i]=1; }
            continue;
        }
        int idx=queue[front++];
        int exec=(rem[idx]<tq)?rem[idx]:tq;
        rem[idx]-=exec; time+=exec;
        for(int i=0; i<n; i++)
            if(!inQ[i] && rem[i]>0 && at[i]<=time)
                { queue[rear++]=i; inQ[i]=1; }
        if(rem[idx]==0) {
            ct[idx]=time; tat[idx]=ct[idx]-at[idx];
            wt[idx]=tat[idx]-bt[idx]; completed++;
        } else queue[rear++]=idx;
    }
    // Re-sort by process number
    for(int i=0; i<n-1; i++)
        for(int j=i+1; j<n; j++)
            if(p[i]>p[j]) {
                swap(p[i],p[j]); swap(at[i],at[j]); swap(bt[i],bt[j]);
                swap(wt[i],wt[j]); swap(tat[i],tat[j]);
            }
    cout << "\nP\tAT\tBT\tWT\tTAT\n";
    for(int i=0; i<n; i++) {
        cout << "P"<<p[i]<<"\t"<<at[i]<<"\t"<<bt[i]<<"\t"<<wt[i]<<"\t"<<tat[i]<<"\n";
        avgwt+=wt[i]; avgtat+=tat[i];
    }
    cout << "\nAverage WT  = " << avgwt/n;
    cout << "\nAverage TAT = " << avgtat/n;
    return 0;
}
